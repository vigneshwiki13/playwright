name: Playwright Tests
on:
  push:
    branches: [ main, master ]
  pull_request:
    branches: [ main, master ]
jobs:
  test:
    timeout-minutes: 60
    runs-on: ubuntu-latest
    steps:
    - uses: actions/checkout@v4                       action: checks out the latest pull or push branch code
    - uses: actions/setup-node@v4                     action: set the environment to run (ubundu latest)
      with:
        node-version: lts/*
    - name: Install dependencies
      run: npm ci                                      shell script: like npm install for to install all dependencies

      run: npx playwright test --config=playwright.service.config.js --workers=4                  step5
